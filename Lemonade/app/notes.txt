// 1. DOMAIN MODELS & ENUMS

/**
 * Representasi state setiap langkah di game Lemonade.
 * Menggunakan Enum menjamin type-safety (tidak akan ada step gaib di luar 4 ini).
 */
enum class LemonadeStep {
    TREE, SQUEEZE, DRINK, RESTART
}

/**
 * Data class untuk membungkus seluruh resource UI per langkah.
 * Ini membuat kode UI bersih dari tumpukan kondisional 'when'.
 */
data class LemonadeUiState(
    val imageRes: Int,
    val contentDescriptionRes: Int,
    val descriptionRes: Int
)

// Extension function untuk memetakan Enum langsung ke Resource UI-nya
fun LemonadeStep.toUiState(): LemonadeUiState = when (this) {
    LemonadeStep.TREE -> LemonadeUiState(
        R.drawable.lemon_tree,
        R.string.lemon_tree_description,
        R.string.lemon_tree
    )
    LemonadeStep.SQUEEZE -> LemonadeUiState(
        R.drawable.lemon_squeeze,
        R.string.lemon_description,
        R.string.lemon
    )
    LemonadeStep.DRINK -> LemonadeUiState(
        R.drawable.lemon_drink,
        R.string.lemon_drink_description,
        R.string.lemon_drink
    )
    LemonadeStep.RESTART -> LemonadeUiState(
        R.drawable.lemon_restart,
        R.string.lemon_empty_description,
        R.string.lemon_empty
    )
}

// ==========================================
// 3. COMPOSABLE SCREENS (STATEFUL)
// ==========================================

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun LemonadeApp() {
    // State Hoisting: Semua state aplikasi ditaruh di level paling atas screen
    var currentStep by remember { mutableStateOf(LemonadeStep.TREE) }
    var targetSqueeze by remember { mutableIntStateOf((2..4).random()) }
    var squeezeCount by remember { mutableIntStateOf(0) }

    // Logika Bisnis dipisah dari fungsi UI murni
    val onStepClick = {
        when (currentStep) {
            LemonadeStep.TREE -> {
                currentStep = LemonadeStep.SQUEEZE
            }
            LemonadeStep.SQUEEZE -> {
                squeezeCount++
                if (squeezeCount >= targetSqueeze) {
                    currentStep = LemonadeStep.DRINK
                }
            }
            LemonadeStep.DRINK -> {
                currentStep = LemonadeStep.RESTART
            }
            LemonadeStep.RESTART -> {
                // Reset semua state saat game diulang
                squeezeCount = 0
                targetSqueeze = (2..4).random()
                currentStep = LemonadeStep.TREE
            }
        }
    }

    Scaffold(
        topBar = {
            CenterAlignedTopAppBar(
                title = { Text(text = "Lemonade", fontWeight = FontWeight.Bold) },
                colors = TopAppBarDefaults.centerAlignedTopAppBarColors(
                    containerColor = MaterialTheme.colorScheme.primaryContainer
                )
            )
        },
        modifier = Modifier.fillMaxSize()
    ) { innerPadding ->
        // Oper state dan event-handler ke bawah (Unidirectional Data Flow)
        LemonadeContent(
            currentStep = currentStep,
            onImageClick = onStepClick,
            modifier = Modifier.padding(innerPadding)
        )
    }
}

// ==========================================
// 4. COMPOSABLE COMPONENTS (STATELESS & REUSABLE)
// ==========================================

/**
 * Komponen UI murni (Stateless). Komponen ini tidak tahu cara mengubah data,
 * dia hanya tahu cara menampilkan data dan meneruskan sinyal klik ke atas.
 */
@Composable
fun LemonadeContent(
    currentStep: LemonadeStep,
    onImageClick: () -> Unit,
    modifier: Modifier = Modifier
) {
    // Dapatkan data resource berdasarkan objek state saat ini
    val uiState = currentStep.toUiState()

    Column(
        modifier = modifier
            .fillMaxSize()
            .wrapContentHeight(),
        horizontalAlignment = Alignment.CenterHorizontally,
    ) {
        Button(
            onClick = onImageClick,
            shape = RoundedCornerShape(dimensionResource(R.dimen.button_corner_radius)),
            colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFD2E7DA))
        ) {
            Image(
                painter = painterResource(uiState.imageRes),
                contentDescription = stringResource(uiState.contentDescriptionRes),
                modifier = Modifier
                    .size(
                        width = dimensionResource(R.dimen.button_image_width),
                        height = dimensionResource(R.dimen.button_image_height)
                    )
                    .padding(dimensionResource(R.dimen.button_interior_padding))
            )
        }
        Spacer(modifier = Modifier.height(16.dp))
        Text(
            text = stringResource(uiState.descriptionRes),
            style = MaterialTheme.typography.bodyLarge
        )
    }
}
